function consultarIA(texto) { const t = texto.toLowerCase().normalize("NFD").replace(/[\u0300-\u036f]/g, ""); const hoy = new Date(); hoy.setHours(0, 0, 0, 0); function fechaProximo(e) { if (!e.fecha) return null; const f = new Date(e.fecha + "T00:00:00"); f.setDate(f.getDate() + Number(e.periodo || 180)); return f; } function fechaTexto(f) { return f ? f.toLocaleDateString("es-CO") : "Sin fecha"; } function diasRestantes(f) { return Math.ceil((f - hoy) / 86400000); } // CONSULTA DE EQUIPOS REGISTRADOS if ( t.includes("equipos registrados") || t.includes("lista de equipos") || t.includes("cuantos equipos") || t.includes("mis equipos") ) { if (!equipos.length) { return "Todavía no hay equipos registrados en BioControl."; } return "📋 EQUIPOS REGISTRADOS\n\n" + equipos.map((e, i) => `${i + 1}. ${e.nombre}\nMarca: ${e.marca || "No registrada"}\nModelo: ${e.modelo || "No registrado"}\nUbicación: ${e.ubicacion || "No registrada"}\nResponsable: ${e.responsable || "No registrado"}` ).join("\n\n"); } // MANTENIMIENTOS VENCIDOS if ( t.includes("vencidos") || t.includes("atrasados") || t.includes("mantenimiento pendiente") ) { const vencidos = equipos.filter(e => { const f = fechaProximo(e); return f && f <= hoy; }); if (!vencidos.length) { return "✅ No se encontraron mantenimientos vencidos en los equipos registrados."; } return "⚠️ MANTENIMIENTOS VENCIDOS\n\n" + vencidos.map(e => `Equipo: ${e.nombre}\nFecha límite: ${fechaTexto(fechaProximo(e))}` ).join("\n\n"); } // PRÓXIMOS MANTENIMIENTOS if ( t.includes("proximos") || t.includes("proximos mantenimientos") || t.includes("por vencer") ) { const proximos = equipos.filter(e => { const f = fechaProximo(e); return f && diasRestantes(f) >= 0 && diasRestantes(f) <= 30; }); if (!proximos.length) { return "No hay mantenimientos programados para los próximos 30 días."; } return "📅 PRÓXIMOS MANTENIMIENTOS\n\n" + proximos.map(e => `Equipo: ${e.nombre}\nFecha programada: ${fechaTexto(fechaProximo(e))}\nDías restantes: ${diasRestantes(fechaProximo(e))}` ).join("\n\n"); } // HISTORIAL DE MANTENIMIENTOS if ( t.includes("historial") || t.includes("mantenimientos realizados") ) { if (!mantenimientos.length) { return "No hay mantenimientos registrados todavía."; } return "🛠️ HISTORIAL DE MANTENIMIENTOS\n\n" + [...mantenimientos].reverse().map((m, i) => `${i + 1}. Equipo: ${m.equipo}\nFecha: ${m.fecha}\nTipo: ${m.tipo}\nTécnico: ${m.tecnico || "No registrado"}\nObservaciones: ${m.observaciones || "Sin observaciones"}` ).join("\n\n"); } // CONSULTAR UN EQUIPO POR SU NOMBRE const equipoEncontrado = equipos.find(e => t.includes(String(e.nombre || "").toLowerCase()) ); if (equipoEncontrado) { const e = equipoEncontrado; const proximo = fechaProximo(e); return `📋 INFORMACIÓN DEL EQUIPO Nombre: ${e.nombre} Tipo: ${e.tipo || "No registrado"} Marca: ${e.marca || "No registrada"} Modelo: ${e.modelo || "No registrado"} Ubicación: ${e.ubicacion || "No registrada"} Responsable: ${e.responsable || "No registrado"} Último mantenimiento: ${e.fecha || "Sin registro"} Periodicidad: ${e.periodo || 180} días Próximo mantenimiento: ${fechaTexto(proximo)} Estado: ${ !proximo ? "Sin fecha de mantenimiento registrada" : proximo <= hoy ? "⚠️ Mantenimiento vencido" : "✅ Mantenimiento dentro del plazo" }`; } // AUTÓCLAVES if (t.includes("autoclave") || t.includes("esteril")) { return `🔧 ORIENTACIÓN TÉCNICA: AUTOCLAVE Posibles comprobaciones iniciales: 1. Verificar la alimentación eléctrica. 2. Revisar el nivel y calidad del agua según el fabricante. 3. Inspeccionar visualmente el empaque de la puerta. 4. Comprobar que el drenaje no esté obstruido. 5. Revisar los mensajes o códigos de error. 6. Consultar el manual del modelo específico. Si no alcanza temperatura o presión, registra el código de error y consulta el procedimiento técnico del fabricante. ⚠️ Nunca abras una cámara presurizada ni intervengas sistemas eléctricos energizados.`; } // UNIDADES ODONTOLÓGICAS if ( t.includes("unidad") || t.includes("odontolog") || t.includes("sillon") ) { return `🦷 ORIENTACIÓN TÉCNICA: UNIDAD ODONTOLÓGICA Comprueba inicialmente: 1. Alimentación eléctrica del equipo. 2. Suministro de aire comprimido. 3. Suministro de agua. 4. Estado visible de las mangueras. 5. Funcionamiento del pedal de control. 6. Sistema de succión. 7. Lámpara de iluminación. 8. Movimientos del sillón. Si el sillón no se mueve, identifica si la falla está en el pedal, alimentación, conexiones o sistema de control. No desmontes componentes eléctricos sin desenergizar el equipo y seguir el manual técnico.`; } // COMPRESORES if (t.includes("compresor") || t.includes("presion")) { return `🔧 ORIENTACIÓN TÉCNICA: COMPRESOR Revisiones iniciales: 1. Verificar alimentación eléctrica. 2. Observar el manómetro. 3. Revisar conexiones externas. 4. Escuchar si existen fugas de aire. 5. Comprobar ventilación. 6. Revisar filtros y drenaje según el fabricante. 7. Verificar si el motor presenta sobrecalentamiento. Si no alcanza la presión requerida, registra la presión observada y el tiempo de funcionamiento. ⚠️ No desmontes recipientes presurizados.`; } // MONITORES Y ELECTROCARDIÓGRAFOS if ( t.includes("monitor") || t.includes("electrocard") || t.includes("ecg") ) { return `🩺 ORIENTACIÓN TÉCNICA: MONITOR BIOMÉDICO Revisa: 1. Alimentación eléctrica. 2. Estado de la batería. 3. Cables y conectores. 4. Sensores y accesorios. 5. Configuración del equipo. 6. Mensajes de alarma. 7. Pruebas funcionales con simulador cuando corresponda. No utilices mediciones dudosas para tomar decisiones clínicas.`; } // FALLAS GENERALES if ( t.includes("falla") || t.includes("no funciona") || t.includes("no enciende") || t.includes("alarma") || t.includes("error") ) { return `🔍 ANÁLISIS INICIAL DE FALLA Para orientar el diagnóstico necesito conocer: 1. Nombre del equipo. 2. Marca y modelo. 3. Síntoma que presenta. 4. Código de alarma o error. 5. Momento en que aparece la falla. 6. Si el equipo tuvo mantenimiento reciente. Con esos datos puedo ayudarte a organizar las posibles causas y las comprobaciones iniciales. ⚠️ La orientación no sustituye el manual del fabricante ni la intervención de personal técnico autorizado.`; } // RESPUESTA GENERAL return `🤖 BIOCONTROL IA - ASISTENTE TÉCNICO Puedo ayudarte con: 📋 Consultar equipos registrados. ⚠️ Identificar mantenimientos vencidos. 📅 Consultar próximos mantenimientos. 🛠️ Revisar el historial. 🦷 Orientar sobre unidades odontológicas. ♨️ Orientar sobre autoclaves. 💨 Orientar sobre compresores. 🩺 Orientar sobre monitores biomédicos. 🔍 Analizar síntomas y posibles fallas. Prueba preguntando: - ¿Qué equipos tengo registrados? - ¿Qué mantenimientos están vencidos? - ¿Qué mantenimientos vienen próximamente? - Muéstrame el historial. - Mi autoclave no alcanza temperatura. - Mi unidad odontológica no tiene presión. Estoy listo para ayudarte.`; }
